// ==========================================
// DB 연동 및 초기 설정
// ==========================================
import { saveGameScore } from "../../common/firebase-config.js";

const canvas = document.getElementById("bounceGame");
const ctx = canvas.getContext("2d");

// ==========================================
// 1. 전역 상태 변수
// ==========================================
let leftPressed = false;
let rightPressed = false;
let currentStage = 0;
let isGameRunning = false;
let animationId;
let cameraX = 0;
let targetCameraX = 0;
let isTransitioning = false;

// ==========================================
// 2. 물리 엔진 환경 변수 
// ==========================================
const boxCount = 20;
const box = canvas.width / boxCount; // 타일 1개의 픽셀 크기 (20px)
const gravity = 0.08;                // 중력 가속도
const jumpForce = -2.2;              // 기본 점프 반발력
const moveSpeed = 1.4;               // 좌우 이동 속도

// 공(Player) 객체 상태
let ball = { x: 0, y: 0, r: 3, dx: 0, dy: 0, trail: [] };

// ==========================================
// 3. 맵 데이터 세팅 (20x20 Grid)
// [Tile Index] 0: 빈공간, 1: 벽, 2: 가시, 3: 슈퍼점프 타일, 4: 출구, 5: 시작점
// ==========================================
const maps = [
	// ... (이곳에 기존 파일의 Stage 1 ~ Stage 10 맵 배열 데이터를 그대로 복사해 넣으세요) ...
];

// ==========================================
// 4. 모달 및 UI DOM 요소 캐싱
// ==========================================
const modal = document.getElementById("gameModal");
const startBtn = document.getElementById("startBtn");
const restartBtn = document.getElementById("restartBtn");
const rankInputArea = document.getElementById("rankInputArea");
const nicknameInput = document.getElementById("nicknameInput");

// ==========================================
// 5. 게임 상태 제어 로직 (시작, 종료, 트랜지션)
// ==========================================
function initStage(stageIdx) {
	const map = maps[stageIdx];
	const yOffset = canvas.height - map.length * box;
	ball.dy = 0;
	ball.trail = [];
	isTransitioning = false;

	// 맵 데이터에서 시작점(5) 위치를 찾아 공의 초기 좌표 세팅
	for (let r = 0; r < map.length; r++) {
		for (let c = 0; c < map[r].length; c++) {
			if (map[r][c] === 5) {
				ball.x = c * box + box / 2 + stageIdx * canvas.width;
				ball.y = r * box + box / 2 + yOffset;
				return;
			}
		}
	}
}

function showStartModal() {
	isGameRunning = false;
	document.getElementById("modalTitle").innerText = "BOUNCE BALL";
	document.getElementById("modalDesc").innerText = "좌우 방향키로 다음 세상으로 나아가세요!";
	rankInputArea.style.display = "none";
	restartBtn.style.display = "none";

	startBtn.innerText = "게임 시작";
	startBtn.disabled = false;

	startBtn.onclick = function () {
		modal.style.display = "none";
		currentStage = 0;
		cameraX = 0;
		targetCameraX = 0;
		initStage(currentStage);
		document.getElementById("stage").innerText = currentStage + 1;
		isGameRunning = true;
		update();
	};
	modal.style.display = "flex";
}

function showGameOverModal(stageScore, isAllClear = false) {
	isGameRunning = false;
	ball.dx = 0;
	cancelAnimationFrame(animationId);

	document.getElementById("modalTitle").innerText = isAllClear ? "ALL CLEAR!" : "GAME OVER";
	document.getElementById("modalDesc").innerText = isAllClear 
		? "축하합니다! 모든 스테이지를 정복했습니다!" 
		: `아쉽네요! 최고 도달: STAGE ${stageScore}`;

	rankInputArea.style.display = "block";
	restartBtn.style.display = "block";
	nicknameInput.value = "";

	startBtn.innerText = "랭킹 등록";
	startBtn.disabled = false;

	// 랭킹 DB 저장 처리
	startBtn.onclick = async () => {
		const nickname = nicknameInput.value.trim();
		if (!nickname || nickname.length < 2) return alert("닉네임을 2자 이상 입력해주세요!");

		startBtn.disabled = true;
		startBtn.innerText = "데이터 확인 중...";

		const result = await saveGameScore("bounce", nickname, stageScore);
		alert(result.msg);

		if (result.status !== "error") {
			showStartModal();
		} else {
			startBtn.disabled = false;
			startBtn.innerText = "랭킹 등록";
		}
	};

	// 다시 시작 처리
	restartBtn.onclick = () => {
		modal.style.display = "none";
		currentStage = 0;
		cameraX = 0;
		targetCameraX = 0;
		initStage(currentStage);
		document.getElementById("stage").innerText = currentStage + 1;
		isGameRunning = true;
		update();
	};

	modal.style.display = "flex";
}

// 다음 스테이지 이동 연출
function startTransition() {
	if (currentStage < maps.length - 1) {
		currentStage++;
		isTransitioning = true;
		targetCameraX += canvas.width; // 카메라 우측 이동

		const nextMap = maps[currentStage];
		const yOff = canvas.height - nextMap.length * box;
		for (let r = 0; r < nextMap.length; r++) {
			for (let c = 0; c < nextMap[r].length; c++) {
				if (nextMap[r][c] === 5) {
					ball.x = c * box + box / 2 + currentStage * canvas.width;
					ball.y = r * box + box / 2 + yOff;
					ball.dx = 0;
					break;
				}
			}
		}
		document.getElementById("stage").innerText = currentStage + 1;
	} else {
		showGameOverModal(10, true);
	}
}

function gameOver() {
	showGameOverModal(currentStage + 1, false);
}

	if (e.key === "ArrowLeft") leftPressed = true;
	if (e.key === "ArrowRight") rightPressed = true;
	if (["ArrowUp", "ArrowDown", "ArrowLeft", "ArrowRight", " "].includes(e.key)) e.preventDefault();
});
window.addEventListener("keyup", (e) => {
	if (e.key === "ArrowLeft") leftPressed = false;
	if (e.key === "ArrowRight") rightPressed = false;
});

const lBtn = document.getElementById("leftBtn");
const rBtn = document.getElementById("rightBtn");
if (lBtn && rBtn) {
	lBtn.onpointerdown = () => (leftPressed = true);
	lBtn.onpointerup = () => (leftPressed = false);
	rBtn.onpointerdown = () => (rightPressed = true);
	rBtn.onpointerup = () => (rightPressed = false);
}

// ==========================================
// 7. 핵심 물리 엔진 & 업데이트 루프
// ==========================================
function update() {
	if (!isGameRunning) return;

	if (!isTransitioning) {
		ball.dy += gravity; // 매 프레임 중력 적용
		
		let inputDx = rightPressed ? moveSpeed : leftPressed ? -moveSpeed : 0;
		let nextX = ball.x + inputDx + ball.dx;
		let nextY = ball.y + ball.dy;
		const map = maps[currentStage];

		// X축(좌우) 벽 충돌 검증
		let hitX = checkWall(nextX, ball.y, map);
		if (!hitX) {
			ball.x = nextX;
			ball.dx *= 0.95; // 마찰력 적용 (자연스러운 감속)
		} else {
			// 벽타기 로직: 벽에 충돌 시 반대 방향으로 강하게 튕김
			if (nextX > ball.x) ball.dx = -moveSpeed * 2.3;
			else ball.dx = moveSpeed * 2.3;
			ball.dy = jumpForce * 0.8;
			ball.x += ball.dx; 
		}

		// Y축(상하) 충돌 검증 및 점프
		let hitY = checkWall(ball.x, nextY, map);
		if (!hitY) {
			ball.y = nextY;
		} else {
			if (ball.dy > 0) {
				// 바닥에 닿았을 때 타일 종류(3=슈퍼점프)에 따라 점프력 차등 적용
				ball.dy = hitY === 3 ? jumpForce * 1.6 : jumpForce;
				ball.dx = 0;
			} else {
				ball.dy = 0; // 천장 충돌 시 상승 정지
			}
		}

		// 잔상(Trail) 효과를 위한 이전 좌표 저장
		ball.trail.push({ x: ball.x, y: ball.y });
		if (ball.trail.length > 12) ball.trail.shift();
		checkTiles(map);
	} else {
		// 스테이지 전환 시 카메라 패닝 효과
		cameraX += (targetCameraX - cameraX) * 0.08;
		if (Math.abs(cameraX - targetCameraX) < 1) {
			cameraX = targetCameraX;
			isTransitioning = false;
		}
	}
	draw();
	animationId = requestAnimationFrame(update);
}

// 충돌 타일 판별 함수: 충돌 시 해당 타일의 번호를 리턴
function checkWall(x, y, map) {
	const yOffset = canvas.height - map.length * box;
	const xOff = currentStage * canvas.width;
	
	// 공의 상하좌우 4포인트를 기준으로 충돌 검사
	const pts = [
		{ x: x - ball.r, y: y - ball.r },
		{ x: x + ball.r, y: y - ball.r },
		{ x: x - ball.r, y: y + ball.r },
		{ x: x + ball.r, y: y + ball.r },
	];

	let hitType = 0;
	for (let p of pts) {
		let gx = Math.floor((p.x - xOff) / box);
		let gy = Math.floor((p.y - yOffset) / box);
		if (gy >= 0 && gy < map.length && gx >= 0 && gx < map[gy].length) {
			let t = map[gy][gx];
			if (t === 1) hitType = 1; // 일반 벽
			if (t === 3) return 3;    // 슈퍼 점프 타일은 즉각 반환 (우선순위)
		}
	}
	return hitType;
}

// 특수 타일(가시, 출구) 밟음 판별 및 낙사 체크
function checkTiles(map) {
	const yOffset = canvas.height - map.length * box;
	const xOff = currentStage * canvas.width;
	let gx = Math.floor((ball.x - xOff) / box);
	let gy = Math.floor((ball.y - yOffset) / box);

	if (gy >= 0 && gy < map.length && gx >= 0 && gx < map[gy].length) {
		let tile = map[gy][gx];
		if (tile === 2) gameOver(); // 가시 밟음
		if (tile === 4 && !isTransitioning) startTransition(); // 출구 도달
	}
	
	if (ball.y > canvas.height + 100) gameOver(); // 바닥 밑으로 추락 (낙사)
}

// ==========================================
// 8. 캔버스 그래픽 렌더링
// ==========================================
function draw() {
	ctx.clearRect(0, 0, canvas.width, canvas.height);
	ctx.save();
	ctx.translate(-cameraX, 0); // 카메라 위치에 따른 캔버스 이동

	// 전체 스테이지 맵 렌더링
	for (let s = 0; s < maps.length; s++) {
		const m = maps[s],
			xOff = s * canvas.width,
			yOff = canvas.height - m.length * box;
			
		for (let r = 0; r < m.length; r++) {
			for (let c = 0; c < m[r].length; c++) {
				let t = m[r][c],
					tx = c * box + xOff,
					ty = r * box + yOff;

				if (t === 1) {
					// 일반 벽돌 렌더링
					ctx.fillStyle = "#a82020";
					ctx.fillRect(tx, ty, box, box);
					ctx.fillStyle = "#444";
					ctx.fillRect(tx + 1, ty + 1, box - 2, box - 2);
				} else if (t === 2) {
					// 가시 함정 렌더링 (삼각형)
					ctx.fillStyle = "#ff4444";
					ctx.beginPath();
					ctx.moveTo(tx, ty + box);
					ctx.lineTo(tx + box / 2, ty);
					ctx.lineTo(tx + box, ty + box);
					ctx.fill();
				} else if (t === 3) {
					// 슈퍼 점프 타일 렌더링
					ctx.fillStyle = "#ffeb3b";
					ctx.fillRect(tx, ty, box, box);
					ctx.fillStyle = "#fbc02d";
					ctx.fillRect(tx + 2, ty + 2, box - 4, box - 4);
					ctx.fillStyle = "#f57f17";
					ctx.fillRect(tx + 4, ty + box / 2 - 2, box - 8, 4);
				}
			}
		}
	}

	// 1. 공의 꼬리(잔상) 렌더링 - 투명도 연산
	ball.trail.forEach((p, i) => {
		ctx.beginPath();
		ctx.arc(p.x, p.y, ball.r, 0, Math.PI * 2);
		ctx.fillStyle = `rgba(142, 36, 170, ${i / ball.trail.length})`;
		ctx.fill();
	});

	// 2. 실제 공(Player) 렌더링
	ctx.fillStyle = "#8e24aa";
	ctx.beginPath();
	ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
	ctx.fill();
	ctx.strokeStyle = "#fff";
	ctx.lineWidth = 1;
	ctx.stroke();
	
	ctx.restore();
}
