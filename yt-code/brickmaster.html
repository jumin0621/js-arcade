// ==========================================
// DB 연동 및 캔버스 초기화

// ✅ 랭킹 저장 함수 임포트 (Firebase 연동)
import { saveGameScore } from "../../common/firebase-config.js";

const canvas = document.getElementById("brickGame");
const ctx = canvas.getContext("2d");

// ==========================================
// 1. 전역 상태 및 물리 설정 관리
// ==========================================
let score = 0;
let currentStage = 0;
let isGameRunning = false;
let animationId = null;

// 패들(유저 조작바) 및 공 초기 설정
const paddleHeight = 10;
let paddleWidth = 90;
let paddleX = (canvas.width - paddleWidth) / 2;
let balls = [];

// 맵 렌더링용 다차원 배열 데이터 (스테이지별 난이도 조정)
const stages = [
	{ rows: 3, cols: 6, color: "#00e5ff" }, // Stage 1
	{ rows: 4, cols: 7, color: "#bb86fc" }, // Stage 2
	{ rows: 5, cols: 8, color: "#ff007b" }, // Stage 3
];

let bricks = [];
let rightPressed = false;
let leftPressed = false;

// DOM 캐싱 (모달 및 UI 제어용)
const modal = document.getElementById("gameModal");
const startBtn = document.getElementById("startBtn");
const restartBtn = document.getElementById("restartBtn");
const rankInputArea = document.getElementById("rankInputArea");
const nicknameInput = document.getElementById("nicknameInput");

// ==========================================
// 2. 게임 초기화 및 스테이지 데이터 리셋
// ==========================================
function initGame() {
	score = 0;
	currentStage = 0;
	updateUI();
	resetStageData();
}

function updateUI() {
	document.getElementById("score").innerText = score;
	document.getElementById("stage").innerText = currentStage + 1;
}

// 다음 스테이지 진입 시 공과 벽돌 배열을 재구성합니다.
function resetStageData() {
	paddleX = (canvas.width - paddleWidth) / 2;
	
	// 다중 공 처리를 위해 공을 객체 배열로 관리합니다.
	balls = [
		{ x: canvas.width / 2, y: canvas.height - 30, dx: 3, dy: -3, radius: 7 },
	];

	const s = stages[currentStage];
	bricks = [];
	// [핵심] 이중 반복문을 통해 벽돌의 행렬(2D Array) 데이터 구조를 생성합니다.
	for (let c = 0; c < s.cols; c++) {
		bricks[c] = [];
		for (let r = 0; r < s.rows; r++) {
			bricks[c][r] = { x: 0, y: 0, status: 1 }; // status 1: 생존, 0: 파괴됨
		}
	}
}

// ==========================================
// 3. 모달 제어 로직 (게임의 흐름 제어)
// ==========================================

// [상태 1] 게임 최초 시작
function showStartModal() {
	isGameRunning = false;
	document.getElementById("modalTitle").innerText = "BRICK MASTER";
	document.getElementById("modalDesc").innerText =
		"좌우 방향키를 이용해 모든 벽돌을 제거하세요!";
	rankInputArea.style.display = "none";
	restartBtn.style.display = "none";

	startBtn.innerText = "게임 시작";
	startBtn.disabled = false;
	startBtn.onclick = () => {
		modal.style.display = "none";
		initGame();
		isGameRunning = true;
		if (animationId) cancelAnimationFrame(animationId);
		draw(); // 렌더링 루프 시작
	};
	modal.style.display = "flex";
}

// [상태 2] 스테이지 클리어
function showStageClearModal() {
	isGameRunning = false;
	document.getElementById("modalTitle").innerText = "STAGE CLEAR!";
	document.getElementById("modalDesc").innerText =
		`다음은 STAGE ${currentStage + 1}입니다.`;
	rankInputArea.style.display = "none";
	restartBtn.style.display = "none";

	startBtn.innerText = "다음 단계로";
	startBtn.disabled = false;
	startBtn.onclick = () => {
		modal.style.display = "none";
		resetStageData();
		isGameRunning = true;
		if (animationId) cancelAnimationFrame(animationId);
		draw();
	};
	modal.style.display = "flex";
}

// [상태 3] 게임 오버 또는 올 클리어 (Firebase 랭킹 등록 연동)
function showGameOverModal(finalScore, isAllClear = false) {
	isGameRunning = false;
	document.getElementById("modalTitle").innerText = isAllClear
		? "ALL CLEAR!"
		: "GAME OVER";
	document.getElementById("modalDesc").innerText = isAllClear
		? `축하합니다! 벽돌을 모두 부쉈습니다!\n최종 점수: ${finalScore}점`
		: `아쉽네요! 최종 점수: ${finalScore}점`;

	rankInputArea.style.display = "block";
	restartBtn.style.display = "block";
	nicknameInput.value = "";

	startBtn.innerText = "랭킹 등록";
	startBtn.disabled = false;

	// Firebase 데이터베이스에 스코어 기록
	startBtn.onclick = async () => {
		const nickname = nicknameInput.value.trim();
		if (!nickname || nickname.length < 2)
			return alert("닉네임을 2자 이상 입력해주세요!");

		startBtn.disabled = true;
		startBtn.innerText = "데이터 확인 중...";

		const result = await saveGameScore("brickmaster", nickname, finalScore);
		alert(result.msg);

		if (result.status !== "error") {
			showStartModal();
		} else {
			startBtn.disabled = false;
			startBtn.innerText = "랭킹 등록";
		}
	};

	restartBtn.onclick = () => {
		modal.style.display = "none";
		initGame();
		isGameRunning = true;
		if (animationId) cancelAnimationFrame(animationId);
		draw();
	};

	modal.style.display = "flex";
}

	else if (e.key == "Left" || e.key == "ArrowLeft") leftPressed = true;

document.addEventListener("keyup", (e) => {
	if (e.key == "Right" || e.key == "ArrowRight") rightPressed = false;
	else if (e.key == "Left" || e.key == "ArrowLeft") leftPressed = false;

const rightBtn = document.getElementById("rightBtn");

leftBtn.onpointerup = () => (leftPressed = false);
leftBtn.onpointerleave = () => (leftPressed = false);
rightBtn.onpointerdown = () => (rightPressed = true);
rightBtn.onpointerup = () => (rightPressed = false);
rightBtn.onpointerleave = () => (rightPressed = false);

// ==========================================
// 5. 핵심 렌더링 및 물리 엔진 (Update Loop)
// ==========================================
function draw() {
	if (!isGameRunning) return;

	ctx.clearRect(0, 0, canvas.width, canvas.height);

	const s = stages[currentStage];
	const bW = canvas.width / s.cols - 10; // 캔버스 너비에 비례한 벽돌 크기 연산
	const bH = 20;

	// [1] 벽돌 렌더링
	bricks.forEach((col, c) => {
		col.forEach((b, r) => {
			if (b.status === 1) {
				let bX = c * (bW + 10) + 5;
				let bY = r * (bH + 10) + 30;
				b.x = bX;
				b.y = bY;
				ctx.fillStyle = s.color;
				ctx.beginPath();
				ctx.roundRect(bX, bY, bW, bH, 5); // 모서리가 둥근 벽돌
				ctx.fill();
			}
		});
	});

	// [2] 공 물리 연산 (다중 공 처리)
	for (let index = balls.length - 1; index >= 0; index--) {
		let ball = balls[index];

		// 공 그리기
		ctx.beginPath();
		ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);
		ctx.fillStyle = "#fff";
		ctx.fill();
		ctx.closePath();

		// 위치 업데이트
		ball.x += ball.dx;
		ball.y += ball.dy;

		// 벽면 충돌 처리 (좌/우/상단 튕기기)
		if (
			ball.x + ball.dx > canvas.width - ball.radius ||
			ball.x + ball.dx < ball.radius
		)
			ball.dx = -ball.dx;
		if (ball.y + ball.dy < ball.radius) ball.dy = -ball.dy;

		// 하단 충돌 처리 (패들 방어 or 추락)
		if (ball.y + ball.dy > canvas.height - ball.radius - paddleHeight) {
			if (ball.x > paddleX && ball.x < paddleX + paddleWidth) {
				// 패들에 맞고 튕겨나감
				ball.dy = -ball.dy;
				
				// 🔥 [핵심 로직] 다중 공 증식 (Multi-ball)
				// 패들에 맞을 때마다 10% 확률로 최대 3개까지 공이 복제됩니다.
				if (Math.random() < 0.1 && balls.length < 3) duplicateBall(ball);
			} else if (ball.y + ball.dy > canvas.height - ball.radius) {
				// 패들을 놓쳐 바닥으로 추락
				balls.splice(index, 1);

				// 모든 공이 떨어지면 게임 오버
				if (balls.length === 0) {
					showGameOverModal(score, false);
					return;
				}
			}
		}

		// 🔥 [핵심 로직] AABB(Axis-Aligned Bounding Box) 충돌 알고리즘
		// 공과 2D 배열 상의 벽돌 간의 교차 판정을 수행합니다.
		bricks.forEach((col) =>
			col.forEach((b) => {
				if (
					b.status === 1 && // 살아있는 벽돌이고,
					ball.x > b.x && ball.x < b.x + bW && // X축이 교차하며,
					ball.y > b.y && ball.y < b.y + bH    // Y축이 교차하면 타격!
				) {
					ball.dy = -ball.dy; // 공 반사
					b.status = 0;       // 벽돌 파괴
					score += 10;
					document.getElementById("score").innerText = score;
					checkStageClear();
				}
			}),
		);
	}

	// [3] 패들 렌더링 및 조작 처리
	ctx.fillStyle = "#00e5ff";
	ctx.fillRect(
		paddleX,
		canvas.height - paddleHeight,
		paddleWidth,
		paddleHeight,
	);

	if (rightPressed && paddleX < canvas.width - paddleWidth) paddleX += 7;
	else if (leftPressed && paddleX > 0) paddleX -= 7;

	// 프레임 루프 지속
	animationId = requestAnimationFrame(draw);
}

// ==========================================
// 6. 이벤트 함수 (공 복제, 스테이지 클리어)
// ==========================================

// 타격된 기존 공의 물리량을 복제하여 반대 방향(dx)으로 튀어나가게 합니다.
function duplicateBall(p) {
	balls.push({ x: p.x, y: p.y, dx: -p.dx, dy: p.dy, radius: p.radius });
}

// 모든 배열의 벽돌 상태(status)가 0인지 검사합니다.
function checkStageClear() {
	if (bricks.every((col) => col.every((b) => b.status === 0))) {
		currentStage++;
		if (currentStage < stages.length) {
			updateUI();
			showStageClearModal(); // 다음 스테이지로 진행
		} else {
			showGameOverModal(score, true); // 모든 스테이지 클리어
		}
	}
}

initGame();
showStartModal();
